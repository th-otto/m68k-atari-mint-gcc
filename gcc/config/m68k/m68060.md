;; The 68060 has two Operand Execution Pipelines (OEPs):
;;   pOEP (primary)   -- executes all instructions
;;   sOEP (secondary) -- executes a subset of simple instructions
;;
;; Up to two instructions per cycle when pairing rules are satisfied.
;; Six dispatch tests must all pass for dual-issue:
;;   Test 1: sOEP instruction words in buffer (not modeled -- fetch effect)
;;   Test 2: Instruction classification compatibility
;;   Test 3: sOEP EA mode restrictions (no indexed w/ base disp, no PC-rel)
;;   Test 4: At most one data memory reference per pair
;;   Test 5: No sOEP.AGU register conflict with pOEP (handled by scheduler)
;;   Test 6: No sOEP.IEE register conflict with pOEP (handled by scheduler)

;; =========================================================================
;; Automata and CPU units
;; =========================================================================

(define_automaton "m68060_oep")
(define_cpu_unit "m68060_pOEP, m68060_sOEP" "m68060_oep")

;; sOEP can only dispatch if pOEP also dispatches (pair with primary).
(final_presence_set "m68060_sOEP" "m68060_pOEP")

;; Single-issue reservation: occupies both pipelines.
(define_reservation "m68060_single" "m68060_pOEP + m68060_sOEP")

;; Memory unit: Test 4 -- at most one data memory access per pair.
(define_automaton "m68060_mem")
(define_cpu_unit "m68060_mem" "m68060_mem")

;; =========================================================================
;; Instruction reservations
;; =========================================================================

;; --- pOEP|sOEP, register-to-register (no memory) --------------------------
;; These can go in either pipeline when no memory is involved.
(define_insn_reservation "m68060_both_00" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
                             shift,lea,moveq_l,mov3q_l,move_l,mvsz,
                             tst,clr,cmp,move")
            (eq_attr "op_mem" "00")))
  "m68060_sOEP | m68060_single")

;; --- pOEP|sOEP, non-indexed memory access (load or store) -----------------
;; Can dual-issue but needs the memory unit.
(define_insn_reservation "m68060_both_mem" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,
                             shift,moveq_l,mov3q_l,move_l,mvsz,
                             tst,clr,cmp,move")
            (eq_attr "op_mem" "10,01")))
  "(m68060_sOEP + m68060_mem) | (m68060_single + m68060_mem)")

;; --- pOEP|sOEP type but indexed EA or read-modify-write -------------------
;; Indexed addressing and RMW (op_mem "11") can't go in sOEP (Test 3).
(define_insn_reservation "m68060_both_indexed" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
                             shift,lea,moveq_l,mov3q_l,move_l,mvsz,
                             tst,clr,cmp,move")
            (eq_attr "op_mem" "i0,0i,11,i1,1i")))
  "m68060_single + m68060_mem")

;; --- pOEP-only: multiply ---------------------------------------------------
(define_insn_reservation "m68060_mul" 2
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "mul_w,mul_l"))
  "m68060_single")

;; --- pOEP-only: divide -----------------------------------------------------
(define_insn_reservation "m68060_div" 30
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "div_w,div_l"))
  "m68060_single")

;; --- pOEP-only: branches and flow control ----------------------------------
(define_insn_reservation "m68060_branch" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "bra,bsr,jmp,jsr,rts"))
  "m68060_single")

;; --- pOEP-only: misc (addx, bit ops, link, nop, pea, unlk, trap) ----------
(define_insn_reservation "m68060_pOEP_only" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "alux_l,bitr,bitrw,link,nop,pea,unlk,trap"))
  "m68060_single")

;; --- pOEP-but-allows-sOEP: Scc and Bcc ------------------------------------
;; These go in pOEP but don't block sOEP from dispatching.
(define_insn_reservation "m68060_scc" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "scc"))
  "m68060_pOEP")

(define_insn_reservation "m68060_bcc" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "bcc"))
  "m68060_pOEP")

;; --- FPU instructions (pOEP, allow sOEP pairing) --------------------------
(define_insn_reservation "m68060_fpu" 3
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "falu,fcmp,fmove,fneg,ftst"))
  "m68060_pOEP")

(define_insn_reservation "m68060_fmul" 3
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "fmul"))
  "m68060_pOEP")

(define_insn_reservation "m68060_fdiv" 20
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "fdiv,fsqrt"))
  "m68060_pOEP")

(define_insn_reservation "m68060_fbcc" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "fbcc"))
  "m68060_single")

;; --- ColdFire IB (should not appear for 68060) ----------------------------
(define_insn_reservation "m68060_ib" 0
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "ib,ignore"))
  "nothing")

;; --- Catch-all for unknown instruction type --------------------------------
;; As type attributes are added to more patterns, fewer instructions
;; will fall through to this catch-all.  Conservative: single-issue.
(define_insn_reservation "m68060_unknown" 1
  (and (eq_attr "cpu" "m68060")
       (eq_attr "type" "unknown"))
  "m68060_single")
